```js
export const MAX_SIDE = 2000;

// Size of an image scaled so its longer side is at most `max`.
export function fitSize(width, height, max = MAX_SIDE) {
  const scale = Math.min(1, max / Math.max(width, height));
  return {
    width: Math.max(1, Math.round(width * scale)),
    height: Math.max(1, Math.round(height * scale)),
  };
}

export function powerOfTwo(n, max = 2048) {
  return Math.min(max, 2 ** Math.round(Math.log2(Math.max(1, n))));
}

// Rectangle (stage pixels) of the image shown "contain" inside the stage.
export function fitView(stageW, stageH, imageW, imageH, pad = 0) {
  const scale = Math.min((stageW - 2 * pad) / imageW, (stageH - 2 * pad) / imageH);
  const w = imageW * scale;
  const h = imageH * scale;
  return { x: (stageW - w) / 2, y: (stageH - h) / 2, w, h };
}

// Zoom `view` by `factor` around stage point (cx, cy). Returns `fit` at zoom <= 1.
export function zoomView(view, fit, cx, cy, factor, maxZoom = 8) {
  const current = view.w / fit.w;
  const next = Math.min(maxZoom, current * factor);
  if (next <= 1) return { ...fit };
  const f = next / current;
  return {
    x: cx - (cx - view.x) * f,
    y: cy - (cy - view.y) * f,
    w: view.w * f,
    h: view.h * f,
  };
}

// Texture repeats [across, down] for `repeats` tiles across the wall.
// Aspects are width / height.
export function tileRepeat(repeats, wallAspect, paperAspect) {
  return [repeats, (repeats * paperAspect) / wallAspect];
}

// Fraction [x, y] of the mural that is visible when it covers the wall.
export function coverScale(wallAspect, paperAspect) {
  return paperAspect > wallAspect
    ? [wallAspect / paperAspect, 1]
    : [1, paperAspect / wallAspect];
}

export function exportName(date) {
  const p = (n) => String(n).padStart(2, '0');
  return `tapeta-${date.getFullYear()}-${p(date.getMonth() + 1)}-${p(date.getDate())}-${p(date.getHours())}${p(date.getMinutes())}.jpg`;
}
```

```js
// Corners order everywhere: top-left, top-right, bottom-right, bottom-left.
export const UNIT = [[0, 0], [1, 0], [1, 1], [0, 1]];

// Gaussian elimination with partial pivoting. Returns null when singular.
function solve(A, b) {
  const n = b.length;
  const M = A.map((row, i) => [...row, b[i]]);
  for (let col = 0; col < n; col++) {
    let pivot = col;
    for (let row = col + 1; row < n; row++) {
      if (Math.abs(M[row][col]) > Math.abs(M[pivot][col])) pivot = row;
    }
    if (Math.abs(M[pivot][col]) < 1e-12) return null;
    [M[col], M[pivot]] = [M[pivot], M[col]];
    for (let row = col + 1; row < n; row++) {
      const f = M[row][col] / M[col][col];
      for (let k = col; k <= n; k++) M[row][k] -= f * M[col][k];
    }
  }
  const x = new Array(n).fill(0);
  for (let row = n - 1; row >= 0; row--) {
    let sum = M[row][n];
    for (let k = row + 1; k < n; k++) sum -= M[row][k] * x[k];
    x[row] = sum / M[row][row];
  }
  return x;
}

// 3x3 matrix (row-major, 9 numbers) mapping each src[i] to dst[i], or null.
export function homography(src, dst) {
  const A = [];
  const b = [];
  for (let i = 0; i < 4; i++) {
    const [x, y] = src[i];
    const [u, v] = dst[i];
    A.push([x, y, 1, 0, 0, 0, -u * x, -u * y]);
    b.push(u);
    A.push([0, 0, 0, x, y, 1, -v * x, -v * y]);
    b.push(v);
  }
  const h = solve(A, b);
  return h ? [...h, 1] : null;
}

export function apply(m, [x, y]) {
  const w = m[6] * x + m[7] * y + m[8];
  return [(m[0] * x + m[1] * y + m[2]) / w, (m[3] * x + m[4] * y + m[5]) / w];
}

export function isConvex(points) {
  let sign = 0;
  for (let i = 0; i < 4; i++) {
    const [ax, ay] = points[i];
    const [bx, by] = points[(i + 1) % 4];
    const [cx, cy] = points[(i + 2) % 4];
    const cross = (bx - ax) * (cy - by) - (by - ay) * (cx - bx);
    if (Math.abs(cross) < 1e-9) return false;
    if (sign === 0) sign = Math.sign(cross);
    else if (Math.sign(cross) !== sign) return false;
  }
  return true;
}

// Estimated wall width / height from the quad, measured in image pixels.
export function wallAspect(corners, width, height) {
  const length = (a, b) => Math.hypot((a[0] - b[0]) * width, (a[1] - b[1]) * height);
  const [tl, tr, br, bl] = corners;
  const w = (length(tl, tr) + length(bl, br)) / 2;
  const h = (length(tl, bl) + length(tr, br)) / 2;
  return w / Math.max(h, 1e-6);
}

// Row-major 3x3 -> column-major array for gl.uniformMatrix3fv.
export function toGL(m) {
  return [m[0], m[3], m[6], m[1], m[4], m[7], m[2], m[5], m[8]];
}
```

```js
// BT.601 YCbCr, all components 0..1. The fragment shader in renderer.js and
// the smart-eraser filter in mask.js use the same formulas; keep them in sync.
export function ycbcr(r, g, b) {
  const R = r / 255, G = g / 255, B = b / 255;
  const y = 0.299 * R + 0.587 * G + 0.114 * B;
  return [y, 0.5 + (B - y) * 0.564, 0.5 + (R - y) * 0.713];
}

export const LUMA_WEIGHT = 0.25;

export function distance(a, b) {
  return Math.hypot(a[1] - b[1], a[2] - b[2]) + LUMA_WEIGHT * Math.abs(a[0] - b[0]);
}

export function median(values) {
  const sorted = [...values].sort((a, b) => a - b);
  return sorted[Math.floor(sorted.length / 2)];
}

const lerp = (a, b, t) => a + (b - a) * t;

// Median [Y, Cb, Cr] of a grid of samples from the central 80% of the quad.
// `image` is { data, width, height } (ImageData). Null when no sample is inside.
export function wallReference(image, corners, grid = 32) {
  const [tl, tr, br, bl] = corners;
  const ys = [], cbs = [], crs = [];
  for (let j = 0; j < grid; j++) {
    for (let i = 0; i < grid; i++) {
      const u = 0.1 + (0.8 * (i + 0.5)) / grid;
      const v = 0.1 + (0.8 * (j + 0.5)) / grid;
      const x = lerp(lerp(tl[0], tr[0], u), lerp(bl[0], br[0], u), v);
      const y = lerp(lerp(tl[1], tr[1], u), lerp(bl[1], br[1], u), v);
      if (x < 0 || x >= 1 || y < 0 || y >= 1) continue;
      const index = (Math.floor(y * image.height) * image.width + Math.floor(x * image.width)) * 4;
      const c = ycbcr(image.data[index], image.data[index + 1], image.data[index + 2]);
      ys.push(c[0]); cbs.push(c[1]); crs.push(c[2]);
    }
  }
  if (ys.length === 0) return null;
  return [median(ys), median(cbs), median(crs)];
}

// Sensitivity 0..1 -> distance threshold of the automatic mask.
// 0 turns the mask off (threshold above any possible distance).
export function threshold(sensitivity) {
  if (sensitivity <= 0) return 10;
  const t = Math.min(1, Math.max(0, (sensitivity - 0.01) / 0.99));
  return 0.30 + (0.03 - 0.30) * t;
}
```

```js
import { LUMA_WEIGHT } from './color.js';

export const NEUTRAL = 128; // automatic mask decides
export const ERASE = 0;     // wallpaper removed
export const RESTORE = 255; // wallpaper forced
export const UNDO_LIMIT = 20;
export const SMART_LIMIT = 0.08;

// Rectangles are inclusive: { x0, y0, x1, y1 }.
export function union(a, b) {
  if (!a) return b;
  if (!b) return a;
  return {
    x0: Math.min(a.x0, b.x0), y0: Math.min(a.y0, b.y0),
    x1: Math.max(a.x1, b.x1), y1: Math.max(a.y1, b.y1),
  };
}

function copyRect(data, width, rect) {
  const w = rect.x1 - rect.x0 + 1;
  const h = rect.y1 - rect.y0 + 1;
  const out = new Uint8Array(w * h);
  for (let row = 0; row < h; row++) {
    const start = (rect.y0 + row) * width + rect.x0;
    out.set(data.subarray(start, start + w), row * w);
  }
  return out;
}

export class Mask {
  constructor(width, height) {
    this.width = width;
    this.height = height;
    this.data = new Uint8Array(width * height).fill(NEUTRAL);
    this.undoStack = [];
    this.snapshot = null;
    this.dirty = null;
  }

  begin() {
    this.snapshot = this.data.slice();
    this.dirty = null;
  }

  // Soft round dab. `filter(index)` may veto single pixels.
  // Returns the touched rectangle, or null when the dab misses the mask.
  dab(cx, cy, radius, value, filter) {
    const x0 = Math.max(0, Math.floor(cx - radius));
    const y0 = Math.max(0, Math.floor(cy - radius));
    const x1 = Math.min(this.width - 1, Math.ceil(cx + radius));
    const y1 = Math.min(this.height - 1, Math.ceil(cy + radius));
    if (x0 > x1 || y0 > y1) return null;
    const soft = Math.max(1, radius * 0.3);
    for (let y = y0; y <= y1; y++) {
      for (let x = x0; x <= x1; x++) {
        const dist = Math.hypot(x - cx, y - cy);
        if (dist > radius) continue;
        const index = y * this.width + x;
        if (filter && !filter(index)) continue;
        const strength = Math.min(1, (radius - dist) / soft);
        this.data[index] = Math.round(this.data[index] + (value - this.data[index]) * strength);
      }
    }
    const rect = { x0, y0, x1, y1 };
    this.dirty = union(this.dirty, rect);
    return rect;
  }

  // Finishes the stroke. Returns its rectangle, or null when nothing was drawn.
  end() {
    const rect = this.dirty;
    if (rect && this.snapshot) {
      this.undoStack.push({ rect, pixels: copyRect(this.snapshot, this.width, rect) });
      if (this.undoStack.length > UNDO_LIMIT) this.undoStack.shift();
    }
    this.snapshot = null;
    this.dirty = null;
    return rect;
  }

  // Drops the stroke in progress. Returns the rectangle that changed back.
  cancel() {
    const rect = this.dirty;
    if (this.snapshot) this.data = this.snapshot;
    this.snapshot = null;
    this.dirty = null;
    return rect;
  }

  // Reverts the last finished stroke. Returns its rectangle, or null.
  undo() {
    const entry = this.undoStack.pop();
    if (!entry) return null;
    const { rect, pixels } = entry;
    const w = rect.x1 - rect.x0 + 1;
    for (let row = 0; row <= rect.y1 - rect.y0; row++) {
      this.data.set(pixels.subarray(row * w, (row + 1) * w), (rect.y0 + row) * this.width + rect.x0);
    }
    return rect;
  }

  patch(rect) {
    return copyRect(this.data, this.width, rect);
  }
}

// Filter for the smart eraser: true for pixels whose color is close to the
// pixel at (cx, cy). Same math as color.ycbcr + color.distance, inlined so the
// hot loop allocates nothing.
export function similarFilter(image, cx, cy, limit = SMART_LIMIT) {
  const { data, width, height } = image;
  const px = Math.min(width - 1, Math.max(0, Math.round(cx)));
  const py = Math.min(height - 1, Math.max(0, Math.round(cy)));
  const at = (index) => {
    const R = data[index * 4] / 255, G = data[index * 4 + 1] / 255, B = data[index * 4 + 2] / 255;
    const y = 0.299 * R + 0.587 * G + 0.114 * B;
    return [y, (B - y) * 0.564, (R - y) * 0.713];
  };
  const [refY, refCb, refCr] = at(py * width + px);
  return (index) => {
    const R = data[index * 4] / 255, G = data[index * 4 + 1] / 255, B = data[index * 4 + 2] / 255;
    const y = 0.299 * R + 0.587 * G + 0.114 * B;
    const cb = (B - y) * 0.564;
    const cr = (R - y) * 0.713;
    return Math.hypot(cb - refCb, cr - refCr) + LUMA_WEIGHT * Math.abs(y - refY) < limit;
  };
}
```

```js
import { homography, UNIT, toGL, wallAspect } from './homography.js';
import { tileRepeat, coverScale, powerOfTwo } from './layout.js';
import { threshold } from './color.js';

const VERTEX = `
attribute vec2 aPos;
varying vec2 vP;
void main() {
  vP = vec2(aPos.x * 0.5 + 0.5, 0.5 - aPos.y * 0.5); // canvas 0..1, y down
  gl_Position = vec4(aPos, 0.0, 1.0);
}`;

// Color math must match src/color.js (ycbcr, distance).
const FRAGMENT = `
precision highp float;
varying vec2 vP;
uniform sampler2D uRoom;
uniform sampler2D uPaper;
uniform sampler2D uMask;
uniform mat3 uH;          // image (0..1) -> wall (0..1)
uniform vec4 uView;       // image rectangle inside the canvas: x, y, w, h (0..1)
uniform vec2 uRepeat;     // tiled: repeats across, down
uniform vec2 uCover;      // mural: visible fraction x, y
uniform float uTiled;
uniform float uPaperOn;   // 0 hides the wallpaper (no paper yet, or "before")
uniform float uThreshold;
uniform float uShadow;
uniform float uBrightness;
uniform vec3 uRef;        // wall reference Y, Cb, Cr
uniform vec3 uBackground;

vec3 ycc(vec3 c) {
  float y = dot(c, vec3(0.299, 0.587, 0.114));
  return vec3(y, 0.5 + (c.b - y) * 0.564, 0.5 + (c.r - y) * 0.713);
}

void main() {
  vec2 uv = (vP - uView.xy) / uView.zw;
  if (uv.x < 0.0 || uv.x > 1.0 || uv.y < 0.0 || uv.y > 1.0) {
    gl_FragColor = vec4(uBackground, 1.0);
    return;
  }
  vec3 room = texture2D(uRoom, uv).rgb;
  vec3 w = uH * vec3(uv, 1.0);
  vec2 wall = w.xy / w.z;
  float edge = 0.002;
  // No test on the sign of w.z: it is negative across the whole quad for
  // strongly foreshortened walls, and the 0..1 range checks are sufficient.
  float inside = smoothstep(0.0, edge, wall.x) * smoothstep(0.0, edge, 1.0 - wall.x)
    * smoothstep(0.0, edge, wall.y) * smoothstep(0.0, edge, 1.0 - wall.y);

  vec3 c = ycc(room);
  float d = length(c.yz - uRef.yz) + 0.25 * abs(c.x - uRef.x);
  float automatic = 1.0 - smoothstep(uThreshold, uThreshold + 0.03, d);
  float m = texture2D(uMask, uv).r; // 0 erase, 0.502 neutral, 1 restore
  float alpha = mix(automatic * clamp(m / 0.502, 0.0, 1.0), 1.0, clamp((m - 0.502) / 0.498, 0.0, 1.0));
  alpha *= inside * uPaperOn;

  vec2 paperUv = uTiled > 0.5
    ? wall * uRepeat
    : clamp((wall - 0.5) * uCover + 0.5, 0.002, 0.998); // REPEAT would bleed the opposite edge
  vec3 paper = texture2D(uPaper, paperUv).rgb;
  float shade = clamp(mix(1.0, c.x / max(uRef.x, 0.05), uShadow), 0.2, 1.6);
  vec3 covered = clamp(paper * uBrightness * shade, 0.0, 1.0);
  gl_FragColor = vec4(mix(room, covered, alpha), 1.0);
}`;

const UNIFORMS = [
  'uRoom', 'uPaper', 'uMask', 'uH', 'uView', 'uRepeat', 'uCover', 'uTiled',
  'uPaperOn', 'uThreshold', 'uShadow', 'uBrightness', 'uRef', 'uBackground',
];
const BACKGROUND = [0.165, 0.157, 0.149]; // #2a2826, same as --stage

// Returns null when WebGL is unavailable. `onRestored` runs after a lost
// context comes back; the caller must upload its images again.
export function createRenderer(canvas, onRestored) {
  const gl = canvas.getContext('webgl', { alpha: false, antialias: false, preserveDrawingBuffer: true });
  if (!gl) return null;

  let uniform = {};
  let textures = {};

  function shader(type, source) {
    const s = gl.createShader(type);
    gl.shaderSource(s, source);
    gl.compileShader(s);
    if (!gl.getShaderParameter(s, gl.COMPILE_STATUS)) throw new Error(gl.getShaderInfoLog(s));
    return s;
  }

  function init() {
    const program = gl.createProgram();
    gl.attachShader(program, shader(gl.VERTEX_SHADER, VERTEX));
    gl.attachShader(program, shader(gl.FRAGMENT_SHADER, FRAGMENT));
    gl.linkProgram(program);
    if (!gl.getProgramParameter(program, gl.LINK_STATUS)) throw new Error(gl.getProgramInfoLog(program));
    gl.useProgram(program);

    gl.bindBuffer(gl.ARRAY_BUFFER, gl.createBuffer());
    gl.bufferData(gl.ARRAY_BUFFER, new Float32Array([-1, -1, 1, -1, -1, 1, 1, 1]), gl.STATIC_DRAW);
    const position = gl.getAttribLocation(program, 'aPos');
    gl.enableVertexAttribArray(position);
    gl.vertexAttribPointer(position, 2, gl.FLOAT, false, 0, 0);

    uniform = {};
    UNIFORMS.forEach((name) => { uniform[name] = gl.getUniformLocation(program, name); });
    textures = { room: gl.createTexture(), paper: gl.createTexture(), mask: gl.createTexture() };
    gl.uniform1i(uniform.uRoom, 0);
    gl.uniform1i(uniform.uPaper, 1);
    gl.uniform1i(uniform.uMask, 2);
    gl.pixelStorei(gl.UNPACK_ALIGNMENT, 1);
    // 1x1 placeholder so drawing without a wallpaper does not sample an unbound texture.
    bind(1, textures.paper);
    gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, 1, 1, 0, gl.RGBA, gl.UNSIGNED_BYTE, new Uint8Array([255, 255, 255, 255]));
  }

  function bind(unit, texture) {
    gl.activeTexture(gl.TEXTURE0 + unit);
    gl.bindTexture(gl.TEXTURE_2D, texture);
  }

  function clampLinear() {
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
  }

  function setRoom(image) {
    bind(0, textures.room);
    gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, image);
    clampLinear();
  }

  // WebGL 1 needs power-of-two textures for REPEAT and mipmaps.
  function setPaper(image) {
    const scaled = document.createElement('canvas');
    scaled.width = powerOfTwo(image.width);
    scaled.height = powerOfTwo(image.height);
    scaled.getContext('2d').drawImage(image, 0, 0, scaled.width, scaled.height);
    bind(1, textures.paper);
    gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, scaled);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.REPEAT);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.REPEAT);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR_MIPMAP_LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
    gl.generateMipmap(gl.TEXTURE_2D);
  }

  function setMask(mask) {
    bind(2, textures.mask);
    gl.texImage2D(gl.TEXTURE_2D, 0, gl.LUMINANCE, mask.width, mask.height, 0, gl.LUMINANCE, gl.UNSIGNED_BYTE, mask.data);
    clampLinear();
  }

  // Uploads only the changed rectangle ({ x0, y0, x1, y1 }, inclusive).
  function updateMask(mask, rect) {
    bind(2, textures.mask);
    gl.texSubImage2D(
      gl.TEXTURE_2D, 0, rect.x0, rect.y0, rect.x1 - rect.x0 + 1, rect.y1 - rect.y0 + 1,
      gl.LUMINANCE, gl.UNSIGNED_BYTE, mask.patch(rect),
    );
  }

  // p: { room, corners, view, hasPaper, showOriginal, tiled, repeats,
  //      paperAspect, aspectFactor, sensitivity, shadow, brightness, ref }
  // `view` is the image rectangle as fractions of the canvas.
  function draw(p) {
    gl.viewport(0, 0, canvas.width, canvas.height);
    if (!p.room || !p.view) {
      gl.clearColor(...BACKGROUND, 1);
      gl.clear(gl.COLOR_BUFFER_BIT);
      return;
    }
    const matrix = homography(p.corners, UNIT) || [0, 0, 0, 0, 0, 0, 0, 0, 0];
    const aspect = wallAspect(p.corners, p.room.width, p.room.height) * p.aspectFactor;
    gl.uniformMatrix3fv(uniform.uH, false, toGL(matrix));
    gl.uniform4f(uniform.uView, p.view.x, p.view.y, p.view.w, p.view.h);
    gl.uniform2fv(uniform.uRepeat, tileRepeat(p.repeats, aspect, p.paperAspect));
    gl.uniform2fv(uniform.uCover, coverScale(aspect, p.paperAspect));
    gl.uniform1f(uniform.uTiled, p.tiled ? 1 : 0);
    gl.uniform1f(uniform.uPaperOn, p.hasPaper && !p.showOriginal ? 1 : 0);
    gl.uniform1f(uniform.uThreshold, threshold(p.sensitivity));
    gl.uniform1f(uniform.uShadow, p.shadow);
    gl.uniform1f(uniform.uBrightness, p.brightness);
    gl.uniform3fv(uniform.uRef, p.ref);
    gl.uniform3fv(uniform.uBackground, BACKGROUND);
    gl.drawArrays(gl.TRIANGLE_STRIP, 0, 4);
  }

  // Renders at the room photo's resolution and returns a 2D canvas copy.
  function exportCanvas(p) {
    const { width, height } = canvas;
    canvas.width = p.room.width;
    canvas.height = p.room.height;
    draw({ ...p, view: { x: 0, y: 0, w: 1, h: 1 }, showOriginal: false });
    const out = document.createElement('canvas');
    out.width = p.room.width;
    out.height = p.room.height;
    out.getContext('2d').drawImage(canvas, 0, 0);
    canvas.width = width;
    canvas.height = height;
    draw(p);
    return out;
  }

  canvas.addEventListener('webglcontextlost', (event) => event.preventDefault());
  canvas.addEventListener('webglcontextrestored', () => { init(); onRestored(); });

  init();
  return { setRoom, setPaper, setMask, updateMask, draw, exportCanvas };
}
```

```js
const LOUPE = 120;      // CSS pixels
const MAGNIFY = 3;
const INSET = 22;       // half of the handle, CSS pixels
const clamp = (value) => Math.min(1.2, Math.max(-0.2, value));

export function setupCorners(ctx) {
  const { state } = ctx;
  const handles = [...document.querySelectorAll('.handle')];
  const outline = document.getElementById('outline');
  const quad = document.getElementById('quad');
  const loupe = document.getElementById('loupe');
  const glass = loupe.getContext('2d');

  function place() {
    const show = state.step === 2 && !!state.room;
    outline.toggleAttribute('hidden', !show); // <svg> has no `hidden` property
    handles.forEach((handle, index) => {
      handle.hidden = !show;
      if (!show) return;
      const [x, y] = ctx.toStage(...state.corners[index]);
      handle.style.transform = `translate(${x}px, ${y}px)`;
    });
    if (show) {
      quad.setAttribute('points', state.corners.map((c) => ctx.toStage(...c).join(',')).join(' '));
    }
  }

  // Magnified view of the photo around image point (u, v), shown above the finger.
  function showLoupe(u, v) {
    const source = (LOUPE / MAGNIFY) * (state.room.width / state.view.w); // photo pixels
    glass.fillStyle = '#2a2826';
    glass.fillRect(0, 0, loupe.width, loupe.height);
    glass.drawImage(
      state.room,
      u * state.room.width - source / 2, v * state.room.height - source / 2, source, source,
      0, 0, loupe.width, loupe.height,
    );
    glass.strokeStyle = '#fff';
    glass.lineWidth = 2;
    glass.beginPath();
    glass.moveTo(loupe.width / 2, loupe.height / 2 - 24);
    glass.lineTo(loupe.width / 2, loupe.height / 2 + 24);
    glass.moveTo(loupe.width / 2 - 24, loupe.height / 2);
    glass.lineTo(loupe.width / 2 + 24, loupe.height / 2);
    glass.stroke();

    const [x, y] = ctx.toStage(u, v);
    const top = y - LOUPE - 40 < 0 ? y + 40 : y - LOUPE - 40;
    loupe.style.transform = `translate(${x - LOUPE / 2}px, ${top}px)`;
    loupe.hidden = false;
  }

  handles.forEach((handle, index) => {
    handle.addEventListener('pointerdown', (event) => {
      handle.setPointerCapture(event.pointerId);
      event.preventDefault();
    });
    handle.addEventListener('pointermove', (event) => {
      if (!handle.hasPointerCapture(event.pointerId)) return;
      // Keep the handle reachable: never let it leave the stage.
      const box = ctx.stage.getBoundingClientRect();
      const x = Math.min(box.right - INSET, Math.max(box.left + INSET, event.clientX));
      const y = Math.min(box.bottom - INSET, Math.max(box.top + INSET, event.clientY));
      const [u, v] = ctx.toImage(x, y);
      const point = [clamp(u), clamp(v)];
      const next = state.corners.map((corner, i) => (i === index ? point : corner));
      if (!isConvex(next)) return; // ignore moves that would fold the quad
      state.corners = next;
      showLoupe(point[0], point[1]);
      ctx.requestDraw();
    });
    const release = () => {
      loupe.hidden = true;
      ctx.updateReference();
      ctx.requestDraw();
    };
    handle.addEventListener('pointerup', release);
    handle.addEventListener('pointercancel', release);
  });

  ctx.onDraw(place);
}
```

```js
import { ERASE, RESTORE, similarFilter, union } from './mask.js';
import { zoomView } from './layout.js';

const $ = (id) => document.getElementById(id);

export function setupPaint(ctx) {
  const { state, stage } = ctx;
  const pointers = new Map(); // pointerId -> { x, y } in stage pixels
  let stroke = null;          // { last, value, filter }
  let pinch = null;           // { cx, cy, d }

  const active = () => state.step === 4 && !!state.mask;

  function stagePoint(event) {
    const box = stage.getBoundingClientRect();
    return { x: event.clientX - box.left, y: event.clientY - box.top };
  }

  function photoPoint(event) {
    const [u, v] = ctx.toImage(event.clientX, event.clientY);
    return [u * state.room.width, v * state.room.height];
  }

  function upload(rect) {
    if (!rect) return;
    ctx.renderer.updateMask(state.mask, rect);
    ctx.requestDraw();
  }

  function dabLine(from, to) {
    const radius = state.brush * Math.max(state.room.width, state.room.height);
    const steps = Math.max(1, Math.ceil(Math.hypot(to[0] - from[0], to[1] - from[1]) / (radius / 3)));
    let rect = null;
    for (let i = 1; i <= steps; i++) {
      const t = i / steps;
      rect = union(rect, state.mask.dab(
        from[0] + (to[0] - from[0]) * t, from[1] + (to[1] - from[1]) * t,
        radius, stroke.value, stroke.filter,
      ));
    }
    upload(rect);
  }

  function pinchState() {
    const [a, b] = [...pointers.values()];
    return { cx: (a.x + b.x) / 2, cy: (a.y + b.y) / 2, d: Math.max(1, Math.hypot(a.x - b.x, a.y - b.y)) };
  }

  function showUndo() {
    $('undo').disabled = !state.mask || state.mask.undoStack.length === 0;
  }

  stage.addEventListener('pointerdown', (event) => {
    if (!active()) return;
    stage.setPointerCapture(event.pointerId);
    pointers.set(event.pointerId, stagePoint(event));
    if (pointers.size === 1) {
      const point = photoPoint(event);
      state.mask.begin();
      stroke = {
        last: point,
        value: state.tool === 'erase' ? ERASE : RESTORE,
        filter: state.smart ? similarFilter(state.roomData, point[0], point[1]) : null,
      };
      dabLine(point, point);
    } else {
      if (stroke) { // a second finger turns the gesture into pan/zoom
        upload(state.mask.cancel());
        stroke = null;
      }
      pinch = pointers.size === 2 ? pinchState() : null;
    }
  });

  stage.addEventListener('pointermove', (event) => {
    if (!pointers.has(event.pointerId)) return;
    pointers.set(event.pointerId, stagePoint(event));
    if (stroke && pointers.size === 1) {
      const point = photoPoint(event);
      dabLine(stroke.last, point);
      stroke.last = point;
    } else if (pinch && pointers.size === 2) {
      const now = pinchState();
      const view = zoomView(state.view, state.fit, now.cx, now.cy, now.d / pinch.d);
      if (view.w > state.fit.w) {
        view.x += now.cx - pinch.cx;
        view.y += now.cy - pinch.cy;
      }
      state.view = view;
      pinch = now;
      ctx.requestDraw();
    }
  });

  function release(event) {
    if (!pointers.delete(event.pointerId)) return;
    if (stroke) {
      state.mask.end();
      stroke = null;
      showUndo();
    }
    pinch = pointers.size === 2 ? pinchState() : null; // third finger lifted: restart from the two left
  }
  stage.addEventListener('pointerup', release);
  stage.addEventListener('pointercancel', release);

  stage.addEventListener('wheel', (event) => {
    if (!active()) return;
    event.preventDefault();
    const { x, y } = stagePoint(event);
    state.view = zoomView(state.view, state.fit, x, y, event.deltaY < 0 ? 1.15 : 1 / 1.15);
    ctx.requestDraw();
  }, { passive: false });

  const toolButtons = [...document.querySelectorAll('#tool button[data-tool]')];
  toolButtons.forEach((button) => {
    button.addEventListener('click', () => {
      state.tool = button.dataset.tool;
      toolButtons.forEach((other) => other.classList.toggle('on', other === button));
    });
  });
  $('smart').addEventListener('change', (event) => { state.smart = event.target.checked; });
  $('brush').addEventListener('input', (event) => { state.brush = Number(event.target.value) / 100; });
  $('undo').addEventListener('click', () => {
    upload(state.mask.undo());
    showUndo();
  });

  ctx.onStep(showUndo);
}
```

```js
import { exportName } from './layout.js';

const $ = (id) => document.getElementById(id);

export function setupOutput(ctx) {
  const { state } = ctx;
  const before = $('before');
  const share = $('share');

  // Hold to see the original photo.
  const show = (original) => {
    state.showOriginal = original;
    ctx.requestDraw();
  };
  before.addEventListener('pointerdown', () => show(true));
  ['pointerup', 'pointercancel', 'pointerleave'].forEach((type) => {
    before.addEventListener(type, () => show(false));
  });

  function render() {
    return new Promise((resolve) => {
      ctx.renderer.exportCanvas(ctx.params()).toBlob(resolve, 'image/jpeg', 0.92);
    });
  }

  $('save').addEventListener('click', async () => {
    const url = URL.createObjectURL(await render());
    const link = document.createElement('a');
    link.href = url;
    link.download = exportName(new Date());
    link.click();
    setTimeout(() => URL.revokeObjectURL(url), 10000);
  });

  // Hide the button where the browser cannot share files.
  const probe = new File([], 'tapeta.jpg', { type: 'image/jpeg' });
  const canShare = !!navigator.canShare && navigator.canShare({ files: [probe] });
  share.hidden = !canShare;
  share.addEventListener('click', async () => {
    const file = new File([await render()], exportName(new Date()), { type: 'image/jpeg' });
    try {
      await navigator.share({ files: [file] });
    } catch (error) {
      if (error.name !== 'AbortError') ctx.toast('Nie udało się udostępnić zdjęcia.');
    }
  });

  // Best effort: mobile browsers do not always ask.
  window.addEventListener('beforeunload', (event) => {
    if (state.room) event.preventDefault();
  });
}
```
